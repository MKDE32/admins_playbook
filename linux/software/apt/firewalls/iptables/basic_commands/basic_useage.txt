# OVERVIEW
Version anzeigen
iptables -V

Listet aktuelle Regeln auf
iptables -L

listet aktuelle Regeln auf erweitert
iptables -vnL



# IPv6
sudo ip6tables -t filter -A INPUT -j DROP



# ESTABLISHED / RELATED
sudo iptables -I INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT



# EXAMPLES
allen input blocken
sudo iptables -t filter -A INPUT -j DROP

SSH zulassen
sudo iptables -I INPUT 1 -p tcp --dport 22 -j ACCEPT

HTTP(S) zulassen
sudo iptables -I INPUT 1 -p tcp --dport 80 -j ACCEPT
sudo iptables -I INPUT 1 -p tcp --dport 443 -j ACCEPT

IP ADRESSE zulassen
sudo iptables -I INPUT 1 -p tcp --dport 22 -s 5.3.6.6 -j ACCEPT



# STANDARTPOLICY
sudo iptables -P INPUT DROP
sudo iptables -P OUTPUT ACCEPT



# BLOCK IP
iptables -A INPUT -s 14.23.59.9 -J DROP
iptables -A INPUT -s 192.168.178.0/24 -J DROP



# LOOPBACK
iptables -A INPUT -i lo -j ACCEPT
iptables -A OUTPUT -i lo -j ACCEPT



# MAC FILTER
-m mac --mac-source 00:08:27:25:7f:20 -j DROP
