```
1. #!/bin/bash
2.  
3. ##** MATCH BY QUOTA **##
4.  
5. iptables -F
6.  
7. # permit max QUOTA bytes from protocol tcp and port 80 (ROUTED TRAFFIC, we are the ROUTER)
8. # traffic is generated by a server inside our LAN. This is downloaded traffic by Internet clients from our server.
9.  
10. ### EDIT the values below ###
11. PROTOCOL="tcp"
12. PORT="80"
13. #1GB
14. QUOTA1="1000000000"
15. #this is the outgoing interface
16. INT="eth0"
17. ###
18.  
19. ###DO NOT EDIT BELOW THIS LINE
20. iptables -A FORWARD -o $INT -p $PROTOCOL --sport $PORT -m quota --quota $QUOTA1 -j ACCEPT
21. iptables -A FORWARD -o $INT -p $PROTOCOL --sport $PORT 				-j DROP
22.  
23.  
24.  
25. # permit max 100MB of incoming https traffic from the web site with the address 100.0.0.1
26.  
27. ### EDIT the values below ###
28. HTTPS_SERVER="100.0.0.1"
29. PROTOCOL="tcp"
30. PORT="443"
31. #100MB
32. QUOTA2="100000000"
33. ###
34.  
35. ###DO NOT EDIT BELOW THIS LINE
36. iptables -A INPUT -s $HTTPS_SERVER -p $PROTOCOL --sport $PORT -m quota --quota $QUOTA2 -j ACCEPT
37. iptables -A INPUT 		   -p $PROTOCOL --sport $PORT 			       -j DROP

```
